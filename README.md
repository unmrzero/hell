    <script>
        // Mobile Navigation Toggle
        const mobileBtn = document.getElementById('mobile-menu-btn');
        const mobileMenu = document.getElementById('mobile-menu');
        
        mobileBtn.addEventListener('click', () => {
            mobileMenu.classList.toggle('hidden');
        });

        // Interactive ROI Calculator Logic
        const billRange = document.getElementById('monthly-bill-range');
        const billValDisplay = document.getElementById('monthly-bill-val');
        const daytimeSelect = document.getElementById('daytime-usage');
        const tariffInput = document.getElementById('tariff-rate');

        const resSystemSize = document.getElementById('res-system-size');
        const resAnnualSaving = document.getElementById('res-annual-saving');
        const resPayback = document.getElementById('res-payback');
        const resCo2 = document.getElementById('res-co2');
        const hiddenCalcSummary = document.getElementById('calc_summary');

        function calculateROI() {
            const monthlyBill = parseFloat(billRange.value);
            const daytimeUsagePercent = parseFloat(daytimeSelect.value);
            const tariff = parseFloat(tariffInput.value) || 4.40;

            billValDisplay.innerText = monthlyBill.toLocaleString('th-TH') + ' บาท';

            // Calculations
            // Monthly units = monthlyBill / tariff
            const monthlyUnits = monthlyBill / tariff;
            const dailyUnits = monthlyUnits / 30;
            const daytimeUnitsNeeded = dailyUnits * daytimeUsagePercent;
            
            // Assuming average 3.9 peak sun hours per day
            const systemSizeKWp = Math.round(daytimeUnitsNeeded / 3.9);
            
            // Turnkey cost estimation (approx 25,000 THB per kWp for commercial scale)
            const estimatedCost = systemSizeKWp * 25000;
            
            // Annual generation & savings
            const annualGeneration = systemSizeKWp * 3.9 * 365;
            const annualSaving = annualGeneration * tariff * daytimeUsagePercent;
            
            // Payback period in years
            const paybackYears = (estimatedCost / annualSaving).toFixed(1);
            
            // CO2 reduction (approx 0.5 kg CO2 per kWh)
            const co2Tons = ((annualGeneration * 0.5) / 1000).toFixed(0);

            // Update DOM
            resSystemSize.innerText = systemSizeKWp.toLocaleString('th-TH') + ' kWp';
            resAnnualSaving.innerText = Math.round(annualSaving).toLocaleString('th-TH') + ' ฿';
            resPayback.innerText = `~ ${paybackYears} ปี`;
            resCo2.innerText = `~ ${co2Tons} ตัน CO2e`;

            // Prepare summary for form submission
            hiddenCalcSummary.value = `[ROI Calc] ค่าไฟ: ${monthlyBill} บ., ระบบ: ${systemSizeKWp}kWp, ประหยัด/ปี: ${Math.round(annualSaving)}บ., คืนทุน: ${paybackYears}ปี`;
        }

        billRange.addEventListener('input', calculateROI);
        daytimeSelect.addEventListener('change', calculateROI);
        tariffInput.addEventListener('input', calculateROI);

        // Initial Calculation Run
        calculateROI();

        // Helper to populate form when clicking ROI button
        function fillFormWithCalcData() {
            const monthlyBill = parseFloat(billRange.value);
            document.getElementById('monthlyElectricityBill').value = monthlyBill.toLocaleString('th-TH') + ' บาท/เดือน';
        }

        // Formspree Submission with Anti-duplicate Safeguards & Conversion Triggers
        const form = document.getElementById('contact-form');
        const submitBtn = document.getElementById('submit-btn');
        const btnText = document.getElementById('btn-text');
        const btnSpinner = document.getElementById('btn-spinner');
        const successAlert = document.getElementById('form-status-success');
        const errorAlert = document.getElementById('form-status-error');

        let isSubmitting = false;

        form.addEventListener('submit', async function(e) {
            e.preventDefault();

            if (isSubmitting) return;

            // UI Loading State
            isSubmitting = true;
            submitBtn.disabled = true;
            btnText.classList.add('hidden');
            btnSpinner.classList.remove('hidden');
            successAlert.classList.add('hidden');
            errorAlert.classList.add('hidden');

            const formData = new FormData(form);

            try {
                const response = await fetch(form.action, {
                    method: 'POST',
                    body: formData,
                    headers: {
                        'Accept': 'application/json'
                    }
                });

                if (response.ok) {
                    // Success State
                    successAlert.classList.remove('hidden');
                    form.reset();
                    
                    // Dispatch Ads/Analytics Events if present
                    if (typeof gtag === 'function') {
                        gtag('event', 'generate_lead', {
                            'event_category': 'Engagement',
                            'event_label': 'LUMINABIU Contact Form Submission'
                        });
                    }
                    if (typeof fbq === 'function') {
                        fbq('track', 'Lead');
                    }

                    // Smooth Scroll to message
                    successAlert.scrollIntoView({ behavior: 'smooth', block: 'center' });
                } else {
                    errorAlert.classList.remove('hidden');
                }
            } catch (error) {
                console.error('Form submission error:', error);
                errorAlert.classList.remove('hidden');
            } finally {
                // Reset Loading State
                isSubmitting = false;
                submitBtn.disabled = false;
                btnText.classList.remove('hidden');
                btnSpinner.classList.add('hidden');
            }
        });
